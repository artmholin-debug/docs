Copy resolves through the localization layer, while data like numbers, keys, and hex codes is left alone. English is the source, and each English string is its own key, so a translation table only lists the strings you change. Anything you skip stays in English, and a regional id like `pt-br` beats the bare `pt`.

With no `locale` given, the window follows the player's Roblox language on its own.

## Translation tables

Pass tables keyed by locale id.

```lua
Rayfield:CreateWindow({
    name = "Example",
    translations = {
        fr = { ["Auto Sprint"] = "Sprint auto", ["Signed in as"] = "Connecté en tant que" },
        es = { ["Auto Sprint"] = "Sprint automático" },
    },
})
```

Switch languages live, with no rebuild.

```lua
window:SetLocale("fr")
```

Add a pack at runtime.

```lua
window:RegisterTranslations({ de = { ["Auto Sprint"] = "Auto-Sprint" } })
```

## A custom resolver

To resolve copy yourself, pass a `translator`. It is a function `(source, localeId)` that returns a string, or `nil` to fall through to the tables. It is checked first, so it wins.

```lua
Rayfield:CreateWindow({
    name = "Example",
    translator = function(source, localeId)
        return myTranslationService(source, localeId)  -- or nil to fall back
    end,
})
```

Swap it at runtime with `window:SetTranslator(fn)`, then re-apply the locale to refresh the screen.

local function AnimateButton(btn, customHoverColor)
    local originalSize = btn.Size
    local originalBg   = btn.BackgroundColor3
    local tweenFast    = TweenInfo.new(0.12, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
    local tweenSlow    = TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)

    local hoverBg = customHoverColor or Color3.new(
        math.min(originalBg.R + 0.15, 1),
        math.min(originalBg.G + 0.15, 1),
        math.min(originalBg.B + 0.15, 1)
    )

    btn.MouseEnter:Connect(function()
        local hoverSize = UDim2.new(originalSize.X.Scale, originalSize.X.Offset + 4, originalSize.Y.Scale, originalSize.Y.Offset + 2)
        TweenService:Create(btn, tweenSlow, {Size = hoverSize, BackgroundColor3 = hoverBg}):Play()
    end)

    btn.MouseLeave:Connect(function()
        TweenService:Create(btn, tweenSlow, {Size = originalSize, BackgroundColor3 = originalBg}):Play()
    end)

    btn.MouseButton1Down:Connect(function()
        local clickSize = UDim2.new(originalSize.X.Scale, originalSize.X.Offset - 2, originalSize.Y.Scale, originalSize.Y.Offset - 2)
        TweenService:Create(btn, tweenFast, {Size = clickSize}):Play()
    end)

    btn.MouseButton1Up:Connect(function()
        local hoverSize = UDim2.new(originalSize.X.Scale, originalSize.X.Offset + 4, originalSize.Y.Scale, originalSize.Y.Offset + 2)
        TweenService:Create(btn, tweenFast, {Size = hoverSize}):Play()
    end)
end

-- Проверка видимости цели за препятствиями (Wall Check)
local function IsVisibleThroughWall(targetPart)
    if not _G.FlingSettings.WallCheckEnabled then
        return true -- Если Wall Check отключён (крестик X), наводимся сквозь любые стены
    end

    local origin      = Camera.CFrame.Position
    local destination = targetPart.Position
    local direction   = destination - origin

    local raycastParams = RaycastParams.new()
    raycastParams.FilterType = RaycastFilterType.Exclude

    local ignoreList = {}
    if player.Character then table.insert(ignoreList, player.Character) end
    if targetPart.Parent then table.insert(ignoreList, targetPart.Parent) end
    raycastParams.FilterDescendantsInstances = ignoreList

    local result = workspace:Raycast(origin, direction, raycastParams)
    return result == nil
end

-- Поиск ближайшей цели внутри круга FOV с учётом Wall Check
local function GetClosestTargetInFOV()
    local closestPlayer    = nil
    local shortestDistance = _G.FlingSettings.FOVSize
    local mousePos         = UserInputService:GetMouseLocation()

    for _, p in ipairs(Players:GetPlayers()) do
        if p ~= player and p.Character then
            local hum        = p.Character:FindFirstChildOfClass("Humanoid")
            local hrp        = p.Character:FindFirstChild("HumanoidRootPart")
            local head       = p.Character:FindFirstChild("Head")
            local targetPart = head or hrp

            if hum and hum.Health > 0 and targetPart then
                if IsVisibleThroughWall(targetPart) then
                    local screenPos, onScreen = Camera:WorldToViewportPoint(targetPart.Position)
                    if onScreen then
                        local dist = (Vector2.new(screenPos.X, screenPos.Y) - mousePos).Magnitude
                        if dist < shortestDistance then
                            shortestDistance = dist
                            closestPlayer    = p
                        end
                    end
                end
            end
        end
    end
    return closestPlayer
end

-- Хук метатаблицы для перехвата выстрелов/наведения (Silent Aim)
local rawmetatable = getrawmetatable or debug.getmetatable
if rawmetatable and setreadonly then
    local gmt = rawmetatable(game)
    setreadonly(gmt, false)
    local oldIndex = gmt.__index

    gmt.__index = newcclosure(function(self, idx)
        if _G.FlingSettings.SilentAimEnabled and not checkcaller() then
            if tostring(idx) == "Hit" or tostring(idx) == "Target" then
                local target = GetClosestTargetInFOV()
                if target and target.Character then
                    local targetPart = target.Character:FindFirstChild("Head") or target.Character:FindFirstChild("HumanoidRootPart")
                    if targetPart then
                        local mouseCFrame  = oldIndex(self, "Hit")
                        local targetCFrame = targetPart.CFrame

                        -- Вычисление параметров Smooth (плавность) и Control (сила перехвата)
                        local controlAlpha = math.clamp(_G.FlingSettings.AimControl / 100, 0.01, 1)
                        local smoothAlpha  = math.clamp(1 / math.max(_G.FlingSettings.AimSmooth, 1), 0.01, 1)
                        local finalAlpha   = controlAlpha * smoothAlpha

                        local interpolatedCFrame = mouseCFrame:Lerp(targetCFrame, finalAlpha)

                        if tostring(idx) == "Hit" then
                            return interpolatedCFrame
                        elseif tostring(idx) == "Target" then
                            return targetPart
                        end
                    end
                end
            end
        end
        return oldIndex(self, idx)
    end)
    setreadonly(gmt, true)
end

RunService.RenderStepped:Connect(function()
    FOVCircle.Position = UserInputService:GetMouseLocation()
    FOVCircle.Visible  = _G.FlingSettings.SilentAimEnabled
    FOVCircle.Radius   = _G.FlingSettings.FOVSize

    local myChar = player.Character
    local myRoot = myChar and myChar:FindFirstChild("HumanoidRootPart")

    for p, esp in pairs(ESP_Cache) do
        local char = p.Character
        local hum  = char and char:FindFirstChildOfClass("Humanoid")
        local hrp  = char and char:FindFirstChild("HumanoidRootPart")

        local isCloseEnough = false
        local distStuds     = 0
        if myRoot and hrp then
            distStuds = (hrp.Position - myRoot.Position).Magnitude
            if distStuds <= _G.FlingSettings.ESPMaxDistance then
                isCloseEnough = true
            end
        end

        if _G.FlingSettings.ESPEnabled and isCloseEnough and char and hum and hrp and hum.Health > 0 then
            local pos, onScreen = Camera:WorldToViewportPoint(hrp.Position)
            if onScreen then
                local head    = char:FindFirstChild("Head")
                local headPos = head and Camera:WorldToViewportPoint(head.Position + Vector3.new(0, 0.5, 0)) or pos
                local legPos  = Camera:WorldToViewportPoint(hrp.Position - Vector3.new(0, 3, 0))

                local height = math.abs(headPos.Y - legPos.Y)
                local width  = height * 0.65
                local x, y   = pos.X - width / 2, pos.Y - height / 2

                local cornerLen = width * 0.25

                -- Отрисовка угловых рам (Corner Box)
                esp.TL1.From = Vector2.new(x, y);             esp.TL1.To = Vector2.new(x + cornerLen, y);             esp.TL1.Visible = true
                esp.TL2.From = Vector2.new(x, y);             esp.TL2.To = Vector2.new(x, y + cornerLen);             esp.TL2.Visible = true

                esp.TR1.From = Vector2.new(x + width, y);     esp.TR1.To = Vector2.new(x + width - cornerLen, y);     esp.TR1.Visible = true
                esp.TR2.From = Vector2.new(x + width, y);     esp.TR2.To = Vector2.new(x + width, y + cornerLen);     esp.TR2.Visible = true

                esp.BL1.From = Vector2.new(x, y + height);     esp.BL1.To = Vector2.new(x + cornerLen, y + height);     esp.BL1.Visible = true
                esp.BL2.From = Vector2.new(x, y + height);     esp.BL2.To = Vector2.new(x, y + height - cornerLen);     esp.BL2.Visible = true

                esp.BR1.From = Vector2.new(x + width, y + height); esp.BR1.To = Vector2.new(x + width - cornerLen, y + height); esp.BR1.Visible = true
                esp.BR2.From = Vector2.new(x + width, y + height); esp.BR2.To = Vector2.new(x + width, y + height - cornerLen); esp.BR2.Visible = true

                -- Шкала здоровья (Health Bar)
                local healthPct = math.clamp(hum.Health / hum.MaxHealth, 0, 1)
                esp.HealthBg.Size     = Vector2.new(3, height)
                esp.HealthBg.Position = Vector2.new(x - 6, y)
                esp.HealthBg.Visible  = _G.FlingSettings.ESPShowHealth

                esp.HealthFill.Size     = Vector2.new(3, height * healthPct)
                esp.HealthFill.Position = Vector2.new(x - 6, y + (height * (1 - healthPct)))
                esp.HealthFill.Color    = Color3.fromHSV(healthPct * 0.3, 1, 1)
                esp.HealthFill.Visible  = _G.FlingSettings.ESPShowHealth

                -- Имя игрока
                esp.Name.Text     = p.Name
                esp.Name.Position = Vector2.new(pos.X, y - 16)
                esp.Name.Visible  = _G.FlingSettings.ESPShowNames

                -- Дистанция
                esp.Distance.Text     = math.floor(distStuds) .. "m"
                esp.Distance.Position = Vector2.new(pos.X, y + height + 4)
                esp.Distance.Visible  = _G.FlingSettings.ESPShowDistance
            else
                for _, obj in pairs(esp) do obj.Visible = false end
            end
        else
            for _, obj in pairs(esp) do obj.Visible = false end
        end
    end
end)

local function ToggleWindow()
    isOpen = not isOpen
    local tInfo = TweenInfo.new(0.25, Enum.EasingStyle.Quart, Enum.EasingDirection.Out)
    if isOpen then
        MainFrame.Visible = true
        TweenService:Create(MainFrame, tInfo, {Size = TARGET_SIZE, BackgroundTransparency = 0}):Play()
    else
        local hideTween = TweenService:Create(MainFrame, tInfo, {Size = UDim2.new(0, 250, 0, 0), BackgroundTransparency = 0.5})
        hideTween:Play()
        hideTween.Completed:Connect(function() if not isOpen then MainFrame.Visible = false end end)
    end
end

local function ResetCharacterPhysics()
    if noclipConnection then noclipConnection:Disconnect(); noclipConnection = nil end
    local char = player.Character
    if char then
        for _, part in ipairs(char:GetChildren()) do if part:IsA("BasePart") then part.CanCollide = true end end
        local hum = char:FindFirstChildOfClass("Humanoid")
        if hum then hum.PlatformStand = false; hum:ChangeState(Enum.HumanoidStateType.GettingUp) end
    end
end

local function ApplyFling()
    if isOpen then ToggleWindow() end
    local char = player.Character
    local root = char and char:FindFirstChild("HumanoidRootPart")
    local hum  = char and char:FindFirstChildOfClass("Humanoid")

    if root and hum then
        hum.PlatformStand = true
        hum:ChangeState(Enum.HumanoidStateType.Physics)
        if noclipConnection then noclipConnection:Disconnect() end

        noclipConnection = RunService.Stepped:Connect(function()
            if char and char.Parent then
                for _, part in ipairs(char:GetChildren()) do if part:IsA("BasePart") then part.CanCollide = false end end
            end
        end)

        local rotForce = math.clamp(_G.FlingSettings.Power / 6, 10, 120)
        root.AssemblyAngularVelocity = Vector3.new(math.random(-rotForce, rotForce), math.random(-rotForce, rotForce), math.random(-rotForce, rotForce))
        root.AssemblyLinearVelocity  = Vector3.new(0, _G.FlingSettings.Power, 0)

        task.delay(math.clamp(_G.FlingSettings.Power / 250, 0.5, 2.2), function() ResetCharacterPhysics() end)
    end
end

UserInputService.InputBegan:Connect(function(input, gameProcessed)
    if gameProcessed then return end

    if isListeningForAimKey then
        local key = (input.UserInputType == Enum.UserInputType.Keyboard and input.KeyCode) or input.UserInputType
        _G.FlingSettings.AimKey = key
        AimKeybindBtn.Text = "AIM KEY: [ " .. tostring(key.Name or key) .. " ]"
        isListeningForAimKey = false
    elseif isListeningForFlingKey and input.UserInputType == Enum.UserInputType.Keyboard then
        _G.FlingSettings.FlingKey = input.KeyCode
        KeybindBtn.Text = "FLING KEY: [ " .. input.KeyCode.Name .. " ]"
        isListeningForFlingKey = false
    elseif isListeningForToggleKey and input.UserInputType == Enum.UserInputType.Keyboard then
        _G.FlingSettings.ToggleKey = input.KeyCode
        ToggleKeybindBtn.Text = "HIDE KEY: [ " .. input.KeyCode.Name .. " ]"
        isListeningForToggleKey = false
    else
        local pressedKey = input.KeyCode ~= Enum.KeyCode.Unknown and input.KeyCode or input.UserInputType
        if pressedKey == _G.FlingSettings.AimKey then
            _G.FlingSettings.SilentAimEnabled = not _G.FlingSettings.SilentAimEnabled
            SilentAimToggleBtn.Text = "SILENT AIM: [ " .. (_G.FlingSettings.SilentAimEnabled and "ON" or "OFF") .. " ]"
        elseif input.KeyCode == _G.FlingSettings.FlingKey then
            ApplyFling()
        elseif input.KeyCode == _G.FlingSettings.ToggleKey then
            ToggleWindow()
        end
    end
end)
